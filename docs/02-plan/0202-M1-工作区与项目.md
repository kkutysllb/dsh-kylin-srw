```ts
/** 最小手写校验：返回错误清单而非抛错（RPC INVALID 聚合用）。 */
export type ValidateFn = (value: unknown) => string[]

export interface ProjectMeta {
  readonly schemaVersion: 1
  readonly slug: string
  readonly title: string
  readonly field?: string
  readonly stage: 'idea' | 'literature' | 'reading' | 'writing' | 'experiments' | 'submission'
  readonly createdAt: string
  readonly updatedAt: string
  readonly targetVenue?: string
  readonly depthTier?: 'D1' | 'D2' | 'D3'
}

export const SLUG_RE = /^[a-z0-9][a-z0-9-]{0,63}$/

export function validateProjectMeta(value: unknown): string[] {
  const errors: string[] = []
  if (typeof value !== 'object' || value === null) return ['project.json 必须是对象']
  const m = value as Record<string, unknown>
  if (m['schemaVersion'] !== 1) errors.push('schemaVersion 必须为 1')
  if (typeof m['slug'] !== 'string' || !SLUG_RE.test(m['slug'])) errors.push('slug 必须匹配 ^[a-z0-9][a-z0-9-]{0,63}$')
  if (typeof m['title'] !== 'string' || m['title'] === '' || m['title'].length > 200) errors.push('title 必填（≤200）')
  const stages = ['idea', 'literature', 'reading', 'writing', 'experiments', 'submission']
  if (typeof m['stage'] !== 'string' || !stages.includes(m['stage'])) errors.push(`stage 必须是 ${stages.join('|')}`)
  for (const key of ['createdAt', 'updatedAt']) {
    if (typeof m[key] !== 'string' || Number.isNaN(Date.parse(m[key] as string))) errors.push(`${key} 必须是 ISO 时间`)
  }
  return errors
}
```

```ts
import { readFile, writeFile } from 'node:fs/promises'
import { rename } from 'node:fs/promises'
import { join } from 'node:path'

export interface RootConfig {
  readonly schemaVersion: 1
  readonly defaultLangPair: 'en-zh' | 'zh-en'
  readonly allowIllustrative: boolean
  readonly offlineMode: boolean
}

export const DEFAULT_ROOT: RootConfig = {
  schemaVersion: 1, defaultLangPair: 'en-zh', allowIllustrative: false, offlineMode: false,
}

export async function readRootConfig(rootAbs: string): Promise<RootConfig> {
  try {
    const raw = JSON.parse(await readFile(join(rootAbs, '.srw', 'root.json'), 'utf8')) as Partial<RootConfig>
    return {
      schemaVersion: 1,
      defaultLangPair: raw['defaultLangPair'] === 'zh-en' ? 'zh-en' : 'en-zh',
      allowIllustrative: raw['allowIllustrative'] === true,
      offlineMode: raw['offlineMode'] === true,
    }
  } catch {
    return DEFAULT_ROOT
  }
}

export async function writeRootConfig(rootAbs: string, patch: Partial<RootConfig>): Promise<RootConfig> {
  const merged = { ...(await readRootConfig(rootAbs)), ...patch, schemaVersion: 1 as const }
  const target = join(rootAbs, '.srw')
  const { mkdir } = await import('node:fs/promises')
  await mkdir(target, { recursive: true })
  const tmp = join(target, `root.json.srw-tmp-${Date.now().toString(36)}`)
  await writeFile(tmp, JSON.stringify(merged, null, 2) + '\n')
  await rename(tmp, join(target, 'root.json'))
  return merged
}
```

- [ ] **Step 1: `src/workspace/project.ts`**——目录即真相源（0102 §6.2）：

```ts
import { readdir, readFile, mkdir, rename, rm, writeFile } from 'node:fs/promises'
import { join } from 'node:path'
import { ServiceError } from '../service.ts'
import { SLUG_RE, validateProjectMeta, type ProjectMeta } from '../core/schema.ts'

export class WorkspaceStore {
  constructor(private readonly rootAbs: () => string | undefined) {}

  private dir(slug: string): string {
    if (!SLUG_RE.test(slug)) throw new ServiceError('INVALID', `slug 非法：${JSON.stringify(slug)}`)
    return join(this.rootAbs() ?? '', slug)
  }

  async list(): Promise<ProjectMeta[]> {
    const root = this.rootAbs()
    if (root === undefined) return []
    const entries = await readdir(root, { withFileTypes: true })
    const projects: ProjectMeta[] = []
    for (const entry of entries) {
      if (!entry.isDirectory() || !SLUG_RE.test(entry.name)) continue
      try {
        const meta = JSON.parse(await readFile(join(root, entry.name, '.srw', 'project.json'), 'utf8')) as ProjectMeta
        if (validateProjectMeta(meta).length === 0) projects.push(meta)
      } catch { /* 无 project.json 的目录不是项目 */ }
    }
    return projects.sort((a, b) => a.createdAt.localeCompare(b.createdAt))
  }

  async create(input: { slug: string; title: string; field?: string }): Promise<ProjectMeta> {
    if (!SLUG_RE.test(input.slug)) throw new ServiceError('INVALID', 'slug 非法')
    const dir = this.dir(input.slug)
    const { stat } = await import('node:fs/promises')
    try { await stat(dir); throw new ServiceError('CONFLICT', `项目目录已存在：${input.slug}`) } catch (e) { if (e instanceof ServiceError) throw e }
    const now = new Date().toISOString()
    const meta: ProjectMeta = {
      schemaVersion: 1, slug: input.slug, title: input.title, field: input.field,
      stage: 'idea', createdAt: now, updatedAt: now,
    }
    for (const sub of ['.srw', 'literature/pdf', 'literature/notes', 'literature/readings', 'paper', 'experiments', 'materials']) {
      await mkdir(join(dir, sub), { recursive: true })
    }
    await writeFile(join(dir, '.srw', 'project.json'), JSON.stringify(meta, null, 2) + '\n')
    return meta
  }

  async open(slug: string): Promise<ProjectMeta> {
    const raw = await readFile(join(this.dir(slug), '.srw', 'project.json'), 'utf8')
    const meta = JSON.parse(raw) as ProjectMeta
    const errors = validateProjectMeta(meta)
    if (errors.length > 0) throw new ServiceError('INVALID', `project.json 损坏：${errors.join('；')}`)
    return meta
  }

  async remove(slug: string): Promise<void> {
    // 二次确认由调用方（RPC 审批/工具 ask）负责；此处只做删除。
    await rm(this.dir(slug), { recursive: true, force: false })
  }

  async rename(slug: string, title: string): Promise<ProjectMeta> {
    const meta = await this.open(slug)
    const next = { ...meta, title, updatedAt: new Date().toISOString() }
    await writeFile(join(this.dir(slug), '.srw', 'project.json'), JSON.stringify(next, null, 2) + '\n')
    return next
  }
}
```
